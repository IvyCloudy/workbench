## 1. 核心概念与架构

### 1.1 VS Code 扩展架构

```
┌─────────────────────────────────────────────────────────┐
│                    VS Code 主进程                        │
├─────────────────────────────────────────────────────────┤
│  ┌─────────────┐  ┌─────────────┐  ┌─────────────────┐│
│  │  Extension  │  │  Extension  │  │    WebView      ││
│  │  Host (Node)│  │  Host (Node)│  │  (隔离的渲染进程) ││
│  │             │  │             │  │                 ││
│  │  • 文件系统  │  │  • 命令注册  │  │  • HTML/CSS/JS  ││
│  │  • 进程管理  │  │  • API 调用  │  │  • 独立 DOM     ││
│  │  • 调试功能  │  │  • 事件监听  │  │  • 消息通信     ││
│  └─────────────┘  └─────────────┘  └─────────────────┘│
│         ↑                    ↑              ↑           │
│    通过 VS Code API    通过 VS Code API   通过 postMessage│
└─────────────────────────────────────────────────────────┘
```

### 1.2 可视化插件的三种形态

| 类型 | 技术栈 | 适用场景 | 复杂度 |
|------|--------|----------|--------|
| **TreeView** | VS Code API | 文件树、列表、导航 | ⭐ |
| **Webview Panel** | HTML/CSS/JS + 框架 | 复杂图表、编辑器、仪表盘 | ⭐⭐⭐ |
| **Custom Editor** | Webview + 二进制处理 | 自定义文件类型编辑器 | ⭐⭐⭐⭐ |

```typescript
import { DashboardPanel } from './webview/panel';

export function activate(context: vscode.ExtensionContext) {
  console.log('Visualization extension activated');

  // 注册命令：打开仪表盘
  const openDashboard = vscode.commands.registerCommand(
    'myVisualization.openDashboard',
    () => {
      DashboardPanel.createOrShow(context.extensionUri);
    }
  );

  // 注册 TreeView
  const treeProvider = new DataTreeProvider();
  vscode.window.createTreeView('myVisualization.treeView', {
    treeDataProvider: treeProvider,
    showCollapseAll: true
  });

  context.subscriptions.push(openDashboard);
}

export function deactivate() {
  console.log('Visualization extension deactivated');
}
```

```typescript
// src/treeview/provider.ts
import * as vscode from 'vscode';

interface TreeNode {
  label: string;
  type: 'folder' | 'file' | 'metric';
  value?: number;
  children?: TreeNode[];
  icon?: string;
}

export class DataTreeProvider implements vscode.TreeDataProvider<TreeNode> {
  private _onDidChangeTreeData = new vscode.EventEmitter<TreeNode | undefined>();
  readonly onDidChangeTreeData = this._onDidChangeTreeData.event;

  private data: TreeNode[] = [
    {
      label: '性能指标',
      type: 'folder',
      children: [
        { label: 'CPU 使用率', type: 'metric', value: 45, icon: 'cpu' },
        { label: '内存占用', type: 'metric', value: 72, icon: 'memory' },
        { label: '响应时间', type: 'metric', value: 120, icon: 'watch' }
      ]
    }
  ];

  getTreeItem(element: TreeNode): vscode.TreeItem {
    const item = new vscode.TreeItem(
      element.label,
      element.type === 'folder' 
        ? vscode.TreeItemCollapsibleState.Collapsed 
        : vscode.TreeItemCollapsibleState.None
    );

    // 动态图标
    if (element.icon) {
      item.iconPath = new vscode.ThemeIcon(element.icon);
    }

    // 显示数值
    if (element.value !== undefined) {
      item.description = `${element.value}${element.label.includes('时间') ? 'ms' : '%'}`;
      
      // 根据数值设置颜色标签
      if (element.value > 80) {
        item.iconPath = new vscode.ThemeIcon('warning', new vscode.ThemeColor('errorForeground'));
      }
    }

    // 点击命令
    if (element.type === 'metric') {
      item.command = {
        command: 'myVisualization.showDetail',
        title: 'Show Detail',
        arguments: [element]
      };
    }

    return item;
  }

  getChildren(element?: TreeNode): TreeNode[] {
    if (!element) return this.data;
    return element.children || [];
  }

  refresh(): void {
    this._onDidChangeTreeData.fire(undefined);
  }
}
```

## 4. WebView 可视化开发

### 4.1 WebView Panel 管理器

```typescript
// src/webview/panel.ts
import * as vscode from 'vscode';
import { getNonce } from '../utils/message';

export class DashboardPanel {
  public static currentPanel: DashboardPanel | undefined;
  private readonly _panel: vscode.WebviewPanel;
  private _disposables: vscode.Disposable[] = [];
  private _extensionUri: vscode.Uri;

  private constructor(panel: vscode.WebviewPanel, extensionUri: vscode.Uri) {
    this._panel = panel;
    this._extensionUri = extensionUri;

    // 设置 HTML 内容
    this._update();

    // 监听面板关闭
    this._panel.onDidDispose(() => this.dispose(), null, this._disposables);

    // 监听消息（从 WebView 到扩展）
    this._panel.webview.onDidReceiveMessage(
      async (message) => {
        switch (message.type) {
          case 'alert':
            vscode.window.showInformationMessage(message.text);
            break;
          case 'fetchData':
            const data = await this._fetchDataFromExtension(message.params);
            this._panel.webview.postMessage({ type: 'data', payload: data });
            break;
          case 'openFile':
            const doc = await vscode.workspace.openTextDocument(message.filePath);
            await vscode.window.showTextDocument(doc);
            break;
        }
      },
      null,
      this._disposables
    );
  }

  public static createOrShow(extensionUri: vscode.Uri): DashboardPanel {
    const column = vscode.window.activeTextEditor
      ? vscode.window.activeTextEditor.viewColumn
      : undefined;

    if (DashboardPanel.currentPanel) {
      DashboardPanel.currentPanel._panel.reveal(column);
      return DashboardPanel.currentPanel;
    }

    const panel = vscode.window.createWebviewPanel(
      'dashboard',
      '数据仪表盘',
      column || vscode.ViewColumn.One,
      {
        enableScripts: true,           // 允许执行 JS
        retainContextWhenHidden: true, // 隐藏时保持状态
        localResourceRoots: [
          vscode.Uri.joinPath(extensionUri, 'media'),
          vscode.Uri.joinPath(extensionUri, 'dist') // 打包后的前端资源
        ]
      }
    );

    DashboardPanel.currentPanel = new DashboardPanel(panel, extensionUri);
    return DashboardPanel.currentPanel;
  }

  private _update() {
    const webview = this._panel.webview;
    this._panel.webview.html = this._getHtmlForWebview(webview);
  }

  private _getHtmlForWebview(webview: vscode.Webview): string {
    const nonce = getNonce();
    
    // 加载本地资源（使用 vscode-resource URI）
    const scriptUri = webview.asWebviewUri(
      vscode.Uri.joinPath(this._extensionUri, 'dist', 'webview.js')
    );
    const styleUri = webview.asWebviewUri(
      vscode.Uri.joinPath(this._extensionUri, 'media', 'style.css')
    );

    return `<!DOCTYPE html>
      <html lang="zh-CN">
      <head>
        <meta charset="UTF-8">
        <meta http-equiv="Content-Security-Policy" 
          content="default-src 'none'; 
                   style-src ${webview.cspSource} 'unsafe-inline'; 
                   script-src 'nonce-${nonce}'; 
                   img-src ${webview.cspSource} https: data:;">
        <meta name="viewport" content="width=device-width, initial-scale=1.0">
        <link href="${styleUri}" rel="stylesheet">
        <title>数据仪表盘</title>
      </head>
      <body>
        <div id="app"></div>
        <script nonce="${nonce}" src="${scriptUri}"></script>
      </body>
      </html>`;
  }

  private async _fetchDataFromExtension(params: any): Promise<any> {
    // 在扩展主机中执行数据获取
    const files = await vscode.workspace.findFiles('**/*.json');
    return files.map(f => ({ name: f.fsPath, size: 0 }));
  }

  public dispose() {
    DashboardPanel.currentPanel = undefined;
    this._panel.dispose();
    while (this._disposables.length) {
      const x = this._disposables.pop();
      if (x) x.dispose();
    }
  }
}
```

### 4.2 消息通信工具

```typescript
// src/utils/message.ts
export function getNonce(): string {
  let text = '';
  const possible = 'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789';
  for (let i = 0; i < 32; i++) {
    text += possible.charAt(Math.floor(Math.random() * possible.length));
  }
  return text;
}

// 类型安全的消息封装
export interface WebviewMessage<T = any> {
  type: string;
  payload: T;
}

// VS Code API 类型声明（用于前端）
declare global {
  interface Window {
    acquireVsCodeApi(): {
      postMessage(message: any): void;
      setState(state: any): void;
      getState(): any;
    };
  }
}
```

```typescript
  VSCodeButton, 
  VSCodeDataGrid, 
  VSCodeDataGridRow, 
  VSCodeDataGridCell,
  VSCodePanels,
  VSCodePanelTab,
  VSCodePanelView
} from '@vscode/webview-ui-toolkit/react';
import { Line } from 'react-chartjs-2';
import {
  Chart as ChartJS,
  CategoryScale,
  LinearScale,
  PointElement,
  LineElement,
  Title,
  Tooltip,
  Legend
} from 'chart.js';

ChartJS.register(CategoryScale, LinearScale, PointElement, LineElement, Title, Tooltip, Legend);

interface AppProps {
  vscode: ReturnType<typeof window.acquireVsCodeApi>;
}

const App: React.FC<AppProps> = ({ vscode }) => {
  const [data, setData] = useState<any[]>([]);
  const [chartData, setChartData] = useState<any>(null);

  useEffect(() => {
    // 监听来自扩展的消息
    const handleMessage = (event: MessageEvent) => {
      const message = event.data;
      switch (message.type) {
        case 'data':
          setData(message.payload);
          updateChart(message.payload);
          break;
        case 'update':
          // 实时推送更新
          setData(prev => [...prev, message.payload]);
          break;
      }

    return () => window.removeEventListener('message', handleMessage);

    setChartData({
      labels: items.map((_, i) => `T-${items.length - i}`),

        label: '性能指标',
        data: items.map(d => d.value),
        borderColor: 'var(--vscode-charts-blue)',
        backgroundColor: 'rgba(0, 120, 212, 0.1)',
        tension: 0.4,
        fill: true
      }]
    });
  };

  const handleRefresh = () => {
    vscode.postMessage({ type: 'fetchData', params: { force: true } });
  };

    vscode.postMessage({ type: 'openFile', filePath });
  };

  return (
    <div style={{ padding: '20px' }}>
      <h2>📊 实时监控仪表盘</h2>
      
      <div style={{ marginBottom: '20px' }}>
        <VSCodeButton onClick={handleRefresh}>
          🔄 刷新数据
        </VSCodeButton>
      </div>

      <VSCodePanels>
        <VSCodePanelTab id="chart">📈 图表视图</VSCodePanelTab>
        <VSCodePanelTab id="table">📋 数据列表</VSCodePanelTab>
        
        <VSCodePanelView id="chart">
          {chartData && (
            <Line 
              data={chartData} 
              options={{
                responsive: true,
                plugins: {
                  legend: { position: 'top' },
                  title: { display: true, text: '实时性能趋势' }
                },
                scales: {
                  y: { 
                    beginAtZero: true,
                    grid: { color: 'var(--vscode-panel-border)' }

                  x: {
                    grid: { color: 'var(--vscode-panel-border)' }
                  }
```

```typescript
// src/editor/customEditor.ts
import * as vscode from 'vscode';
import { DashboardPanel } from '../webview/panel';

export class JsonVisualEditor implements vscode.CustomTextEditorProvider {
  public static readonly viewType = 'myVisualization.jsonEditor';

  constructor(private readonly context: vscode.ExtensionContext) {}

  async resolveCustomTextEditor(
    document: vscode.TextDocument,
    webviewPanel: vscode.WebviewPanel,
    _token: vscode.CancellationToken
  ): Promise<void> {
    const webview = webviewPanel.webview;
    webview.options = { enableScripts: true };
    
    // 监听文档变化
    const changeDocumentSubscription = vscode.workspace.onDidChangeTextDocument(e => {
      if (e.document.uri.toString() === document.uri.toString()) {
        this._updateWebview(webview, document);
      }
    });

    // 监听 WebView 消息（保存等操作）
    webview.onDidReceiveMessage(async (message) => {
      switch (message.type) {
        case 'save':
          const edit = new vscode.WorkspaceEdit();
          edit.replace(
            document.uri,
            new vscode.Range(0, 0, document.lineCount, 0),
            JSON.stringify(message.data, null, 2)
          );
          await vscode.workspace.applyEdit(edit);
          await document.save();
          break;
      }
    });

    webviewPanel.onDidDispose(() => {
      changeDocumentSubscription.dispose();
    });

    this._updateWebview(webview, document);
  }

  private _updateWebview(webview: vscode.Webview, document: vscode.TextDocument) {
    try {
      const json = JSON.parse(document.getText());
      webview.postMessage({
        type: 'init',
        data: json,
        fileName: document.fileName
      });
    } catch {
      webview.postMessage({ type: 'error', message: 'Invalid JSON' });
    }
  }
}

// 在 extension.ts 中注册
export function activate(context: vscode.ExtensionContext) {
  context.subscriptions.push(
    vscode.window.registerCustomEditorProvider(
      JsonVisualEditor.viewType,
      new JsonVisualEditor(context),
      { webviewOptions: { retainContextWhenHidden: true } }
    )
  );
}
```

### 6.1 代码复杂度可视化

```typescript
// 分析代码并生成可视化报告
async function analyzeComplexity(uri: vscode.Uri, webview: vscode.Webview) {
  const document = await vscode.workspace.openTextDocument(uri);
  const text = document.getText();
  
  // 简单复杂度分析
  const lines = text.split('\n');
  const metrics = {
    totalLines: lines.length,
    codeLines: lines.filter(l => l.trim() && !l.trim().startsWith('//')).length,
    commentLines: lines.filter(l => l.trim().startsWith('//') || l.includes('/*')).length,
    functions: (text.match(/function\s+\w+|const\s+\w+\s*=\s*\(.*\)\s*=>/g) || []).length,
    complexity: 0
  };

  // 计算圈复杂度（简化版）
  const branches = text.match(/if|else|for|while|switch|catch|\?/g) || [];
  metrics.complexity = branches.length + 1;

  webview.postMessage({
    type: 'complexity',
    payload: metrics
  });
}
```

```typescript
async function getGitHistory(workspaceRoot: string) {
  const git = simpleGit(workspaceRoot);
```
